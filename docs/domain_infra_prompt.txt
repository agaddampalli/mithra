from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime, timezone


@dataclass
class DomainEvent:
    """Cross-domain side effect emitted after tool execution.

    Events fire AFTER autonomy confirmation — never before.
    If user rejects a confirmation, no events fire.

    ONE-WAY DOOR: events are immutable after creation.
    """
    event_type: str             # e.g. "appointment_created"
    source_domain: str          # e.g. "time"
    payload: dict               # event-specific data
    user_id: str
    household_id: str
    request_id: str             # links to originating request
    timestamp: str = field(
        default_factory=lambda: datetime.now(
            timezone.utc).isoformat())


@dataclass
class ToolResult:
    """Standard result returned by all domain tools.

    LLM uses summary to generate response text.
    Events are emitted by AutononomyEngine after confirmation.

    ONE-WAY DOOR: summary is always present.
    Even on failure: "Could not create event: Google not connected"
    """
    success: bool
    summary: str                # REQUIRED: LLM uses this for response
    data: dict | None = None    # structured result data
    error: str | None = None    # human-readable error
    error_code: str | None = None  # machine-readable e.g. "NOT_CONNECTED"
    undo_token: str | None = None  # if reversible, token for undo
    events: list[DomainEvent] = field(default_factory=list)

    @classmethod
    def ok(
        cls,
        summary: str,
        data: dict | None = None,
        events: list[DomainEvent] | None = None,
        undo_token: str | None = None,
    ) -> "ToolResult":
        """Create a successful result."""
        return cls(
            success=True,
            summary=summary,
            data=data or {},
            events=events or [],
            undo_token=undo_token,
        )

    @classmethod
    def fail(
        cls,
        summary: str,
        error_code: str | None = None,
        error: str | None = None,
    ) -> "ToolResult":
        """Create a failed result."""
        return cls(
            success=False,
            summary=summary,
            error=error or summary,
            error_code=error_code,
        )

from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
from typing import Any


class PermissionTier(Enum):
    """Tool permission level.

    ONE-WAY DOOR: DESTRUCTIVE always requires user confirmation.
    No autonomy override can bypass DESTRUCTIVE tier.
    This is a security guarantee, not a preference.
    """
    READ = "read"               # always execute, no confirm needed
    WRITE = "write"             # execute per autonomy config
    DESTRUCTIVE = "destructive" # ALWAYS confirm, no override possible


@dataclass
class ToolMetadata:
    """Runtime metadata for a domain tool.

    Separate from the @tool function itself so tool logic
    remains clean. AutononomyEngine reads this at execution time.

    name must match the @tool name exactly.
    """
    name: str                       # matches @tool name
    domain_id: str                  # owning domain
    permission: PermissionTier      # read | write | destructive
    is_reversible: bool             # supports undo operation
    autonomy_level: int = 1         # default confirmation level
                                    # 1=SURFACE, 2=ACT_TELL, 3=ACT_SILENT
    event_emitted: str | None = None  # event type after success


@dataclass
class ToolBundle:
    """A LangChain tool paired with Mithra runtime metadata.

    tool: the @tool decorated function for LLM invocation
    metadata: autonomy config, permissions, event emission

    Domain returns list[ToolBundle] from get_tools().
    LangGraph extracts tools for bind_tools().
    AutononomyEngine reads metadata at execution time.
    """
    tool: Any                   # LangChain BaseTool
    metadata: ToolMetadata

from __future__ import annotations
from dataclasses import dataclass, field


@dataclass
class DashboardItem:
    """Single item displayed on HA Lovelace dashboard."""
    id: str
    title: str
    subtitle: str | None = None
    icon: str = "mdi:information"   # HA MDI icon name
    priority: str = "normal"         # "high" | "normal" | "low"
    action: str | None = None        # tool name to call on tap


@dataclass
class DashboardData:
    """Standardized dashboard response for all domains.

    GET /api/v1/dashboard/{domain_id} returns this.
    HA REST sensor polls every 5 minutes.
    summary is the sensor value_template target.
    """
    domain: str
    updated_at: str             # ISO8601
    summary: str                # "2 events today"
    items: list[DashboardItem] = field(default_factory=list)
    alerts: list[str] = field(default_factory=list)
    quick_actions: list[dict] = field(default_factory=list)

    def to_dict(self) -> dict:
        return {
            "domain": self.domain,
            "updated_at": self.updated_at,
            "summary": self.summary,
            "items": [
                {
                    "id": item.id,
                    "title": item.title,
                    "subtitle": item.subtitle,
                    "icon": item.icon,
                    "priority": item.priority,
                    "action": item.action,
                }
                for item in self.items
            ],
            "alerts": self.alerts,
            "quick_actions": self.quick_actions,
        }

from __future__ import annotations
from abc import ABC, abstractmethod
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from mithra.domains.context import DomainContext
    from mithra.domains.bundle import ToolBundle
    from mithra.domains.result import DomainEvent
    from mithra.domains.dashboard import DashboardData


class Domain(ABC):
    """Abstract base class for all Mithra domains.

    Every domain implements this interface.
    Tools are pure functions — no state stored in domain.
    Cross-domain communication via domain events only.
    Domains never import other domain modules.

    ONE-WAY DOOR: this interface. Changing it requires
    updating all domains simultaneously.

    MCP migration: Domain wraps into MCPServerAdapter.
    A2A migration: Domain becomes A2A specialist agent.
    Both migrations require zero changes to domain logic.
    """

    @property
    @abstractmethod
    def domain_id(self) -> str:
        """Unique identifier e.g. 'time', 'people', 'money'.
        Immutable. Used as StorageAdapter domain key.
        Matches domain naming convention in section 7."""

    @property
    @abstractmethod
    def version(self) -> str:
        """Semantic version e.g. '1.0.0'.
        Major bump = breaking tool interface change.
        Required for safe product-scale updates."""

    @property
    @abstractmethod
    def display_name(self) -> str:
        """Human-readable name e.g. 'Time & Calendar'"""

    @abstractmethod
    def get_tools(
        self,
        context: "DomainContext",
    ) -> list["ToolBundle"]:
        """Return tools bound to this request context.

        Tools are context-bound at creation.
        Max 8 tools per domain (prevents context bloat).
        Each tool must use Pydantic args_schema.
        Each tool must return ToolResult as dict.
        """

    @abstractmethod
    async def get_dashboard_data(
        self,
        user_id: str,
        household_id: str,
    ) -> "DashboardData":
        """Standardized summary for HA Lovelace dashboard.
        Called every 5 minutes. Must be fast (<500ms).
        Return empty DashboardData if no data available."""

    def get_context_keys(self) -> list[str]:
        """Memory domains to load for context assembly.
        Override if domain needs cross-domain memory.
        Default: [self.domain_id]"""
        return [self.domain_id]

    def get_event_subscriptions(self) -> list[str]:
        """Domain events this domain reacts to.
        e.g. MoneyDomain returns ['appointment_created']
        Default: [] (no subscriptions)"""
        return []

    async def handle_event(
        self,
        event: "DomainEvent",
        context: "DomainContext",
    ) -> None:
        """React to events from other domains.

        Called AFTER autonomy confirmation only.
        If user rejects confirmation, this is never called.
        Default: no-op.
        """
        pass

    def health(self) -> dict:
        """Domain health status for monitoring."""
        return {
            "domain_id": self.domain_id,
            "version": self.version,
            "status": "ok",
        }

from __future__ import annotations
import logging
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from mithra.domains.base import Domain
    from mithra.domains.context import DomainContext
    from mithra.domains.bundle import ToolBundle

logger = logging.getLogger(__name__)


class DomainRegistry:
    """Central registry for all Mithra domains.

    Bootstrap populates this at startup.
    LangGraph agent queries it per request.
    AutononomyEngine uses it to look up tool metadata.
    """

    def __init__(self) -> None:
        self._domains: dict[str, "Domain"] = {}
        self._tool_metadata_index: dict[str, object] = {}

    def register(self, domain: "Domain") -> None:
        """Register a domain. Called once at bootstrap."""
        self._domains[domain.domain_id] = domain
        logger.info(
            f"Registered domain: {domain.domain_id} "
            f"v{domain.version} "
            f"({domain.display_name})")

    def get(self, domain_id: str) -> "Domain | None":
        """Get domain by id."""
        return self._domains.get(domain_id)

    def get_all_domain_ids(self) -> list[str]:
        """Return all registered domain ids."""
        return list(self._domains.keys())

    def get_tools(
        self,
        domain_ids: list[str],
        context: "DomainContext",
    ) -> list["ToolBundle"]:
        """Return ToolBundles for requested domains.

        Used by LangGraph to bind tools per request.
        Indexes metadata for AutononomyEngine lookups.
        """
        bundles: list["ToolBundle"] = []
        for domain_id in domain_ids:
            domain = self._domains.get(domain_id)
            if not domain:
                logger.warning(
                    f"Domain not found: {domain_id}")
                continue
            domain_bundles = domain.get_tools(context)
            for bundle in domain_bundles:
                # Index by tool name for fast lookup
                self._tool_metadata_index[
                    bundle.tool.name] = bundle.metadata
            bundles.extend(domain_bundles)
        return bundles

    def get_tool_metadata(
        self, tool_name: str
    ) -> object | None:
        """Get ToolMetadata by tool name.
        Called by AutononomyEngine at execution time."""
        return self._tool_metadata_index.get(tool_name)

    def get_event_subscribers(
        self, event_type: str
    ) -> list["Domain"]:
        """Return domains subscribed to this event type."""
        return [
            domain
            for domain in self._domains.values()
            if event_type in domain.get_event_subscriptions()
        ]

    def health(self) -> dict:
        """Health of all registered domains."""
        return {
            "registered_domains": len(self._domains),
            "domain_ids": self.get_all_domain_ids(),
            "domains": {
                domain_id: domain.health()
                for domain_id, domain in self._domains.items()
            },
        }

    def _extract_domain_ids(
        self, intent: str
    ) -> list[str]:
        """Extract domain ids from classified intent.

        intent format: "time.calendar_create_event" or "time"
        Returns: ["time"]

        Multi-domain: "time,money" returns ["time", "money"]
        Unknown intent: returns [] (no tools loaded)
        """
        if not intent or "." not in intent and "," not in intent:
            # Check if intent prefix is a known domain
            parts = intent.split(".")
            if parts:
                return [parts[0]]
        # Multi-domain: "time,money"
        if "," in intent:
            return [d.strip() for d in intent.split(",")]
        # Standard: "time.calendar_create_event"
        return [intent.split(".")[0]]
